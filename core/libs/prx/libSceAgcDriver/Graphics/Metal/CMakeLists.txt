cmake_minimum_required(VERSION 3.24)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(AnyPS5Metal LANGUAGES CXX)
    if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
        set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
    endif()
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    include(CTest)
endif()

get_filename_component(ANYPS5_METAL_LIBS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
get_filename_component(ANYPS5_METAL_RECOMPILER_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../shader/recompiler" ABSOLUTE)

add_library(agc_metal_windows STATIC GuestWindows.cpp)
target_include_directories(agc_metal_windows PUBLIC ${ANYPS5_METAL_LIBS_DIR} ${ANYPS5_METAL_RECOMPILER_DIR})
target_compile_features(agc_metal_windows PUBLIC cxx_std_20)
set_target_properties(agc_metal_windows PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)

if(APPLE)
    enable_language(OBJCXX)
    add_library(agc_metal STATIC Device.mm)
    target_link_libraries(agc_metal PUBLIC agc_metal_windows "-framework Metal" "-framework Foundation")
    target_compile_options(agc_metal PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
    set_target_properties(agc_metal PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED ON OBJCXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
    if(TARGET libc)
        target_sources(agc_metal PRIVATE GuestRegistry.cpp)
        target_link_libraries(agc_metal PUBLIC libc)
    endif()
endif()

if(BUILD_TESTING)
    add_executable(agc_metal_guest_windows_tests tests/GuestWindowsTest.cpp)
    target_link_libraries(agc_metal_guest_windows_tests PRIVATE agc_metal_windows)
    add_test(NAME agc_metal_guest_windows COMMAND agc_metal_guest_windows_tests)
    set_tests_properties(agc_metal_guest_windows PROPERTIES TIMEOUT 20)
    if(APPLE)
        add_executable(agc_metal_guest_memory_probe tests/GuestMemoryProbe.mm)
        target_link_libraries(agc_metal_guest_memory_probe PRIVATE agc_metal)
        target_compile_options(agc_metal_guest_memory_probe PRIVATE -fobjc-arc)
        set_target_properties(agc_metal_guest_memory_probe PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED ON OBJCXX_EXTENSIONS OFF)
        add_test(NAME agc_metal_guest_memory_probe COMMAND agc_metal_guest_memory_probe)
        set_tests_properties(agc_metal_guest_memory_probe PROPERTIES TIMEOUT 120)
    endif()
endif()
